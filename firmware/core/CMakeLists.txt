cmake_minimum_required(VERSION 3.20)
set(IDF_TARGET esp32p4)
set(EXTRA_COMPONENT_DIRS "${CMAKE_CURRENT_LIST_DIR}/../../esp32/panel_settings")
set(TRANSPORT_CORE ON)
set(RELEASE_URL "https://transport.pickering.org" CACHE STRING "Public release origin")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/VERSION")
file(READ "${CMAKE_CURRENT_LIST_DIR}/VERSION" release)
string(STRIP "${release}" release)
if(NOT release MATCHES "^(0|[1-9][0-9]*)[.](0|[1-9][0-9]*)[.](0|[1-9][0-9]*)$")
    message(FATAL_ERROR "Invalid core VERSION: ${release}")
endif()
set(major "${CMAKE_MATCH_1}")
set(minor "${CMAKE_MATCH_2}")
set(patch "${CMAKE_MATCH_3}")
if(major GREATER 2147 OR minor GREATER 999 OR patch GREATER 999)
    message(FATAL_ERROR "Core VERSION exceeds the OTA field limits")
endif()
math(EXPR code "${major} * 1000000 + ${minor} * 1000 + ${patch}")
if(code LESS 9000 OR code GREATER 2147483646)
    message(FATAL_ERROR "Core VERSION outside the OTA range")
endif()
# Previously released tablets require this decimal code in the ESP-IDF image header.
set(PROJECT_VER "${code}" CACHE STRING "OTA code from VERSION" FORCE)
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(transport_display)
